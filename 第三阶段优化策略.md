### 1.3 组合优化公式

```python
def optimize_portfolio(expected_returns, cov_matrix, risk_aversion=2.0):
    """
    均值-方差最优化组合
    
    目标: 最大化 组合收益 - λ × 组合风险
    
    参数:
    - expected_returns: 各策略预期收益
    - cov_matrix: 协方差矩阵
    - risk_aversion: 风险厌恶系数
    
    返回: 最优权重分配
    """
    n = len(expected_returns)
    
    # 目标函数系数
    # 最大化收益，最小化风险
    from scipy.optimize import minimize
    
    def objective(weights):
        portfolio_return = np.dot(weights, expected_returns)
        portfolio_var = np.dot(weights.T, np.dot(cov_matrix, weights))
        return -(portfolio_return - risk_aversion * portfolio_var)
    
    # 约束条件
    constraints = [
        {'type': 'eq', 'fun': lambda w: np.sum(w) - 1}  # 权重和=1
    ]
    
    # 边界
    bounds = tuple((0, 1) for _ in range(n))
    
    # 初始值
    x0 = np.ones(n) / n
    
    result = minimize(objective, x0, method='SLSQP', 
                     bounds=bounds, constraints=constraints)
    
    return result.x
```

def calculate_equity_allocation(equity_budget, etf_scores):
    """
    权益类内部配置
    
    公式: 各ETF权重 = 权重因子 × 得分因子

    total_score = sum(etf_scores.values())

        allocations[etf] = equity_budget * (score / total_score)

        allocations[etf] = min(allocations[etf], 0.30)  # 单ETF不超过30%
    
    return allocations

## 三、风险平价策略

### 3.1 风险平价公式

```python
def risk_parity_allocation(returns_dict, target_vol=0.15):
    """
    风险平价配置
    
    目标: 各资产对组合的风险贡献相等
    
    公式:
    Risk_Contribution_i = Weight_i × (Cov(Asset_i, Portfolio) / Portfolio_Vol)
    
    约束: Risk_Contribution_1 = Risk_Contribution_2 = ... = Risk_Contribution_n
    """
    assets = list(returns_dict.keys())
    returns_df = pd.DataFrame(returns_dict)
    
    # 计算协方差矩阵
    cov_matrix = returns_df.cov() * 252  # 年化
    
    def risk_contribution(weights, cov):
        portfolio_vol = np.sqrt(np.dot(weights.T, np.dot(cov, weights)))
        marginal_contrib = np.dot(cov, weights)
        risk_contrib = weights * marginal_contrib / portfolio_vol
        return risk_contrib
    
    def objective(weights):
        target = np.ones(len(assets)) / len(assets)
        actual = risk_contribution(weights, cov_matrix.values)
        return np.sum((actual - target) ** 2)
    
    # 优化求解
    from scipy.optimize import minimize
    
    n = len(assets)
    constraints = {'type': 'eq', 'fun': lambda w: np.sum(w) - 1}
    bounds = tuple((0.05, 0.50) for _ in range(n))
    x0 = np.ones(n) / n
    
    result = minimize(objective, x0, method='SLSQP', 
                     bounds=bounds, constraints=constraints)
    
    weights = dict(zip(assets, result.x))
    
    # 调整到目标波动率
    portfolio_vol = calculate_portfolio_vol(weights, cov_matrix)
    scale_factor = target_vol / portfolio_vol
    
    return {k: v * scale_factor for k, v in weights.items()}
```

class MultiFactorModel:
    """多因子选股模型"""
    
    def __init__(self):
        self.factors = [
            'value_pe', 'value_pb',
            'growth_revenue', 'growth_earning',
            'quality_roe', 'quality_debt',
            'momentum_20', 'momentum_60',
            'volatility_20',
            'sentiment_turnover',
            'money_flow'
        ]
        
    def calculate_factor_scores(self, stock_data):
        """
        计算各因子得分(标准化到0-1)
        """
        scores = {}
        
        for factor in self.factors:
            values = stock_data[factor]
            
            # 市值中性化
            mkt_cap = stock_data['market_cap']
            residual = self._neutralize(values, mkt_cap)
            
            # 标准化
            z_score = (residual - residual.mean()) / residual.std()
            
            # 转换为0-1得分
            scores[factor] = (z_score + 3) / 6
            scores[factor] = scores[factor].clip(0, 1)
        
        return scores
    
    def calculate_composite_score(self, factor_scores, factor_weights=None):
        """
        计算综合得分
        
        公式: Composite = Σ(Factor_i × Weight_i)
        """
        if factor_weights is None:
            factor_weights = self._get_default_weights()
        
        composite = sum([
            factor_weights[f] * factor_scores[f] 
            for f in self.factors
        ])
        
        return composite
    
    def _neutralize(self, values, market_cap):
        """市值中性化"""
        from sklearn.linear_model import LinearRegression
        
        X = np.log(market_cap).values.reshape(-1, 1)
        y = values.values
        
        model = LinearRegression()
        model.fit(X, y)
        
        return y - model.predict(X)
    
    def _get_default_weights(self):
        """默认因子权重"""
        return {
            'value_pe': 0.10,
            'value_pb': 0.05,
            'growth_revenue': 0.10,
            'growth_earning': 0.10,
            'quality_roe': 0.15,
            'quality_debt': 0.05,
            'momentum_20': 0.15,
            'momentum_60': 0.10,
            'volatility_20': -0.10,  # 负权重(低波动优)
            'sentiment_turnover': 0.05,
            'money_flow': 0.15
        }

def event_signal(event_type, days_since_event, intensity):
    """
    事件驱动信号
    
    参数:
    - event_type: 事件类型
    - days_since_event: 距事件天数
    - intensity: 事件强度(0-1)
    
    返回: 信号强度
    """
    
    # 事件生命周期曲线
    def event_curve(days, peak_day, decay_rate):
        """事件冲击曲线"""
        return np.exp(-decay_rate * abs(days - peak_day))
    
    # 各事件参数
    event_params = {
        'earnings': {'peak': 3, 'decay': 0.15},
        'policy': {'peak': 5, 'decay': 0.08},
        'north_flow': {'peak': 2, 'decay': 0.20},
        'breakout': {'peak': 1, 'decay': 0.30},
        'index_adjust': {'peak': 5, 'decay': 0.10}
    }
    
    params = event_params.get(event_type, {'peak': 3, 'decay': 0.15})
    
    signal = event_curve(days_since_event, params['peak'], params['decay'])
    signal = signal * intensity
    
    return signal
```

def performance_attribution(portfolio_returns, benchmark_returns, factors):
    """
    绩效归因分析
    
    分解总收益来源
    """
    
    # 1. 选股收益(Alpha)
    selection_effect = portfolio_returns - calculate_factor_returns(portfolio_returns, factors)
    
    # 2. 因子暴露收益(Beta)
    factor_returns = {}
    for factor in factors:
        factor_returns[factor] = calculate_factor_returns(portfolio_returns, [factor])
    
    # 3. 交互效应
    interaction_effect = portfolio_returns - selection_effect - sum(factor_returns.values())
    
    # 汇总
    attribution = {
        '总收益': portfolio_returns,
        '选股效应(Alpha)': selection_effect,
        '因子暴露(Beta)': factor_returns,
        '交互效应': interaction_effect
    }
    
    return attribution


def calculate_factor_returns(returns, factor_exposure):
    """
    计算因子收益贡献
    
    公式: Factor_Return = Σ(Exposure_i × Factor_i_Return)
    """
    return sum([
        exposure * factor_return 
        for exposure, factor_return in zip(factor_exposure, factor_returns)
    ])
```

class EmergencyHandler:
    """应急事件处理"""
    
    def handle_extreme_event(self, event_type, current_positions):
        """
        处理极端事件
        """
        if event_type == 'market_crash':
            # 市场闪崩
            return self._handle_market_crash(current_positions)
        
        elif event_type == 'liquidity_crisis':
            # 流动性危机
            return self._handle_liquidity_crisis(current_positions)
        
        elif event_type == 'model_failure':
            # 模型失效
            return self._handle_model_failure()

        """市场闪崩处理"""

            'action': 'buy',
            'amount': '30%',
